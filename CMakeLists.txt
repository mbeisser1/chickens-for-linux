# Chickens for Linux — Allegro 4 game (not Allegro 5).
#
# Build (needs network the first time: downloads Allegro 4.4.3.1, then builds it):
#   cmake -S . -B build -DCMAKE_BUILD_TYPE=Release
#   cmake --build build -j
#
# Debug (game only; Allegro is still built as Release by ExternalProject below):
#   cmake -S . -B build-debug -DCMAKE_BUILD_TYPE=Debug
#   cmake --build build-debug -j
#
# Run from the repo root so ./assets/ (dat/, sound/, options.cfg, HighScores) resolve:
#   ./build/chickens4linux
#
# Allegro is built out-of-tree by CMake’s ExternalProject (Allegro’s own CMake
# must not be add_subdirectory’d from here).

cmake_minimum_required(VERSION 3.17)
project(ChickensForLinux LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

include(ExternalProject)

set(_allegro_url
    "https://github.com/liballeg/allegro5/releases/download/4.4.3.1/allegro-4.4.3.1.tar.gz")
set(_alleg_install "${CMAKE_CURRENT_BINARY_DIR}/allegro4-install")
set(_alleg_ep "${CMAKE_CURRENT_BINARY_DIR}/allegro4-ep")
set(_alleg_lib "${_alleg_install}/lib/liballeg.so")

# Allegro is third-party: these flags apply only to the ExternalProject sub-build,
# not to chickens4linux (see target_compile_options below).

if(MSVC)
    set(_alleg_c_flags "/W0")
    set(_alleg_cxx_flags "/W0")
else()
    set(_alleg_c_flags "-w")
    set(_alleg_cxx_flags "-w")
endif()

ExternalProject_Add(chickens_allegro4
    PREFIX "${_alleg_ep}"
    INSTALL_DIR "${_alleg_install}"
    URL "${_allegro_url}"
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE
    CMAKE_ARGS
        -DCMAKE_BUILD_TYPE=Release
        -DCMAKE_INSTALL_PREFIX:PATH=<INSTALL_DIR>
        "-DCMAKE_C_FLAGS=${_alleg_c_flags}"
        "-DCMAKE_CXX_FLAGS=${_alleg_cxx_flags}"
        -DSTRICT_WARN=OFF
        # Docs are generated by a separate target; install can run without those
        # files present and fail. We only need headers and liballeg.
        -DWANT_DOCS=OFF
        -DWANT_EXAMPLES=OFF
        -DWANT_TOOLS=OFF
        -DWANT_TESTS=OFF
    BUILD_BYPRODUCTS "${_alleg_lib}"
)

add_executable(chickens4linux
    src/main.cpp
    src/game.cpp
    src/game_flow.cpp
    src/graphics_display.cpp
    src/asset_manager.cpp
    src/animation.cpp
    src/chicken.cpp
    src/gem.cpp
    src/gore.cpp
    src/helper.cpp
    src/game_state.cpp
    src/highscore.cpp
    src/level.cpp
    src/settings.cpp
    src/smoke.cpp
)

if(MSVC)
    target_compile_options(chickens4linux PRIVATE /W4)
elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(chickens4linux PRIVATE -Wall -Wextra)
endif()

add_dependencies(chickens4linux chickens_allegro4)
target_include_directories(chickens4linux PRIVATE
    "${CMAKE_CURRENT_SOURCE_DIR}/src"
    "${_alleg_install}/include"
)
target_link_libraries(chickens4linux PRIVATE "${_alleg_lib}")
set_target_properties(chickens4linux PROPERTIES
    BUILD_RPATH "${_alleg_install}/lib"
)
